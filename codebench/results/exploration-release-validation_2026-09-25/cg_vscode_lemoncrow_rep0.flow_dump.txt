export function stringifyJsonWithBufferRefs<T>(obj: T, replacer: JSONStringifyReplacer | null = null, useSafeStringify = false): StringifiedJsonWithBufferRefs {
	const foundBuffers: VSBuffer[] = [];
	const serialized = (useSafeStringify ? safeStringify : JSON.stringify)(obj, (key, value) => {
		if (typeof value === 'undefined') {
			return undefinedRef; // JSON.stringify normally converts 'undefined' to 'null'
		} else if (typeof value === 'object') {
			if (value instanceof VSBuffer) {
				const bufferIndex = foundBuffers.push(value) - 1;
				return { [refSymbolName]: bufferIndex };
			}
			if (replacer) {
				return replacer(key, value);
			}
		}
		return value;
	});
	return {
		jsonString: serialized,
		referencedBuffers: foundBuffers
	};
}

[assistant] The extension host runs as a **separate process** and talks to the main renderer over an **RPC layer built on a MessagePort**, not direct calls.

The pipeline (local process ext host):

1. **Transport** — `localProcessExtensionHost.ts` forks the ext host process, then `acquirePort` / `MessagePortExtHostConnection` establishes an Electron `MessagePort`. This is wrapped as an `IMessagePassingProtocol` (`postMessage`/`onMessage` of `Uint8Array` frames) after a handshake (`_establishProtocol`/`_performHandshake`).

2. **RPC** — that protocol is handed to `RPCProtocol` (`rpcProtocol.ts`), which multiplexes request/reply/cancel messages, tracks pending replies via `LazyPromise`, and monitors responsiveness. Messages are serialized with `stringifyJsonWithBufferRefs` (JSON + out-of-band `VSBuffer` refs, URI transforming).

3. **Proxies** — each side registers actors by `ProxyIdentifier`. `getProxy(id)` returns a `Proxied<T>` whose method calls become RPC requests; `set(id, instance)` registers the local handler. Convention: `MainThread*` services live in the renderer, `ExtHost*` in the ext host, wired through `extHostContext` (`extensionHostManager.ts`).

So an extension calling an API → `ExtHost*` proxy → RPCProtocol → MessagePort → renderer's `MainThread*` customer, and vice-versa. Async, buffer-aware, cancellable.
