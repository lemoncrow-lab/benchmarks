    /// Used to represent whether a validation succeeded or failed.

    struct MIMEType {
        let type: String
        let subtype: String

        var isWildcard: Bool { type == "*" && subtype == "*" }

        init?(_ string: String) {
            let components: [String] = {
                let stripped = string.trimmingCharacters(in: .whitespacesAndNewlines)
                let split = stripped[..<(stripped.range(of: ";")?.lowerBound ?? stripped.endIndex)]

                return split.components(separatedBy: "/")
            }()

            if components.count == 2, let type = components.first, let subtype = components.last {
                self.type = type
                self.subtype = subtype
            } else {
                return nil
            }
        }

        func matches(_ mime: MIMEType) -> Bool {
            switch (type, subtype) {
            case (mime.type, mime.subtype), (mime.type, "*"), ("*", mime.subtype), ("*", "*"):
                true
            default:
                false
            }
        }
    }

    // MARK: Properties

    fileprivate var acceptableStatusCodes: Range<Int> { 200..<300 }

    fileprivate var acceptableContentTypes: [String] {
        if let accept = request?.value(forHTTPHeaderField: "Accept") {
            return accept.components(separatedBy: ",")
        }

        return ["*/*"]
    }

    // MARK: Status Code

    fileprivate func validate<S: Sequence>(statusCode acceptableStatusCodes: S,
                                           response: HTTPURLResponse)
        -> ValidationResult
        where S.Iterator.Element == Int {
        if acceptableStatusCodes.contains(response.statusCode) {
            return .success(())
        } else {
            let reason: ErrorReason = .unacceptableStatusCode(code: response.statusCode)
            return .failure(AFError.responseValidationFailed(reason: reason))
        }
    }

    // MARK: Content Type

    fileprivate func validate<S: Sequence>(contentType acceptableContentTypes: S,
                                           response: HTTPURLResponse,
                                           isEmpty: Bool)
        -> ValidationResult
        where S.Iterator.Element == String {
        guard !isEmpty else { return .success(()) }

        return validate(contentType: acceptableContentTypes, response: response)
    }

    fileprivate func validate<S: Sequence>(contentType acceptableContentTypes: S,
                                           response: HTTPURLResponse)
        -> ValidationResult
        where S.Iterator.Element == String {
        guard
            let responseContentType = response.mimeType,
            let responseMIMEType = MIMEType(responseContentType)
        else {
            for contentType in acceptableContentTypes {
                if let mimeType = MIMEType(contentType), mimeType.isWildcard {
                    return .success(())
                }
            }

            let error: AFError = {
                let reason: ErrorReason = .missingContentType(acceptableContentTypes: acceptableContentTypes.sorted())
                return AFError.responseValidationFailed(reason: reason)
            }()

            return .failure(error)
        }

        for contentType in acceptableContentTypes {
            if let acceptableMIMEType = MIMEType(contentType), acceptableMIMEType.matches(responseMIMEType) {

            return AFError.responseValidationFailed(reason: reason)
        }()

        return .failure(error)

[assistant] Alamofire splits this into three phases, all coordinated by `Session` and the `Request` state machine.

## 1. Build

`AF.request(...)` (any `Session.request`/`download`/`upload` overload in `Source/Core/Session.swift`) doesn't build a `URLRequest` immediately. It wraps your inputs in a `URLRequestConvertible` (e.g. `RequestEncodableConvertible`, which holds URL, `HTTPMethod`, `Encodable` parameters, a `ParameterEncoder`, headers, and an optional `RequestModifier`) and constructs the appropriate `Request` subclass (`DataRequest`, `DownloadRequest`, `UploadRequest`, …), then returns it. `performEagerlyIfNecessary` kicks off work if `requestSetup == .eager`.

The actual `URLRequest` is built later in `performSetupOperations` (Session.swift ~L1246), on the `requestQueue`:
1. `convertible.asURLRequest()` — encodes parameters, applies headers and any `requestModifier`.
2. `initialRequest.validate()` — sanity-checks the built request (e.g. GET with a body).
3. `request.didCreateInitialURLRequest(...)` fires event-monitor callbacks.
4. If an adapter exists (`adapter(for:)` merges session- and request-level `RequestInterceptor`s), `adapter.adapt(...)` transforms the request (auth headers, etc.), and the adapted result is re-validated.

## 2. Send

`didCreateURLRequest` (Session.swift ~L1291) runs on `rootQueue`:
- `request.task(for:using:)` creates the underlying `URLSessionTask`.
- The task is stored in `requestTaskMap` and `request.didCreateTask(task)` fires.

`SessionDelegate` receives `URLSession` callbacks and routes them back through `SessionStateProvider`/`RequestDelegate`. When response handlers are added, the request resumes; failures flow through `retryResult`/`retryRequest`, which can re-`perform` the request (with `forRetry: true`).

## 3. Validate

Validation is **opt-in and deferred** (`Source/Features/Validation.swift`). Calling `.validate()` on a `DataRequest` appends a `Validation` closure to the request; it runs when the response arrives, before your response serializer sees the result.

- **Status code**: `acceptableStatusCodes` defaults to `200..<300`; a miss returns `AFError.responseValidationFailed(.unacceptableStatusCode(...))`.
- **Content type**: defaults to the `Accept` header (or `*/*`). `MIMEType.matches` handles wildcards; failures give `.unacceptableContentType` or `.missingContentType`. Empty bodies short-circuit to success.
- Bare `.validate()` chains both: `validate(statusCode:).validate(contentType:)`.

Each validation produces a `ValidationResult` (`Result<Void, Error>`); the first failure is attached to the request and surfaces as the `.failure` in every subsequent response handler.
